## 点击皮肤：已拥有的直接装备；未解锁的先弹确认 —⟦—⟧⟪\n⟫## 一次误触就扣掉几十钻石是不可接受的，价格与余额都要在确认框里写清楚⟦。⟧⟪\n⟫func _on_skin_chip_pressed(skin_id: String) -> void:

func _open_skin_confirm(skin_id: String) -> void:

## 皮肤列表⟦，⟧⟪*⟫*横向排满一行才换行**⟦。⟧⟪\n⟫##
## 列数取「当前宽度一行塞得下几张」与「实际条目数」的较小⟦值⟧⟪␠⟫—⟦—⟧⟪\n⟫## 前者保证只在铺满时才换行，后者保证条目少的时候正好排满一整行⟦。⟧⟪\n⟫## 卡片本身⟦是⟧⟪␠⟫`SIZE_EXPAND_FILL`，列宽因此均分，每行都精确铺满内容区⟦。⟧⟪\n⟫##
## 可用宽度取自 **ScrollContainer 的视⟦口⟧⟪*⟫*，不能取 grid 自己⟦的⟧⟪␠⟫size —⟦—⟧⟪\n⟫## GridContainer 在内容比容器宽时会撑到自己的最小宽度，用它会形成循环依赖⟦。⟧⟪\n⟫func _relayout_skins() -> void:
    var width := _skin_area_width()
    if width <= 1.0:
        return
    for key in ["TableSkinList", "CardSkinList"]:
        var grid := node_named(key) as GridContainer
        if grid == null:
            continue
        var count := grid.get_child_count()
        if count <= 0:
            continue
        var span := SkinChip.MIN_SIZE.x + SKIN_SEPARATION
        var fit := int(floor((width + SKIN_SEPARATION) / span))
        var columns := clampi(mini(fit, count), 1, count)
        if grid.columns != columns:
            grid.columns = columns
    # 网格列数定下来之后各条目的预览区尺寸才确定，烘焙要等这一⟦切⟧⟪\n⟫    _sync_preview_textures()
    _schedule_bake()

# ---------------------------------------------------------------- 预览烘焙
#
# 皮肤预览⟦是⟧⟪*⟫*静⟦态⟧⟪*⟫*的，却每帧都要重画几百条绘制指令（实测整⟦页⟧⟪␠⟫12700 条，
# 而同一页的底噪⟦才⟧⟪␠⟫2286 条），页面因此卡成幻灯片⟦。⟧⟪\n⟫# 这里把每条预览烘成一张贴图，之后每帧只剩一个四边形⟦。⟧⟪\n⟫
## 把缓存里已有的预览贴图发给各条目（命中的条目立刻降到一次绘制指令）
func _sync_preview_textures() -> void:
    for chip in _table_chips + _card_chips:
        var entry := chip as SkinChip
        if entry != null:
            entry.preview_texture = _preview_cache.get(_preview_key(entry))

func _preview_key(chip: SkinChip) -> String:
    var area := chip.preview_size()
    return "%s@%dx%d" % [str(chip.skin.get("id", "")), int(area.x), int(area.y)]

## 把还没烘过的预览**一次⟦性⟧⟪*⟫*烘出来（`PreviewBaker` 会把它们排进同一个离屏视口，
## 只等一帧、只读一次，界面上是"一起出⟦现⟧⟪"⟫而不⟦是⟧⟪"⟫一格一格地蹦出⟦来⟧⟪"⟫）⟦。⟧⟪"⟫##
## 回填⟦时⟧⟪*⟫*按条目对象直接给**，而不是再⟦拿⟧⟪␠⟫key 去查一遍：首次布局里网格列数会变，
## 条目尺寸在同一帧内还会再翻一次，隔一帧再⟦按⟧⟪␠⟫key 查就会全部落⟦空⟧⟪\n⟫## （实测：缓存⟦里⟧⟪␠⟫11 张都在，条目上却一张都没贴上）⟦。⟧⟪\n⟫func _schedule_bake() -> void:
    if _baking or not is_inside_tree():
        return
    _baking = true
    # 让过一帧再量尺寸：此刻拿到的是过渡值，烘出来立刻就作废
    await get_tree().process_frame
    if not is_inside_tree():
        _baking = false
        return
    var requests: Array = []
    for chip in _table_chips + _card_chips:
        var entry := chip as SkinChip
        if entry == null or entry.skin.is_empty():
            continue
        var area := entry.preview_size()
        if area.x < 16.0 or area.y < 16.0:
            continue
        var key := _preview_key(entry)
        if _preview_cache.has(key):
            entry.preview_texture = _preview_cache[key]
            continue
        requests.append({
            "key": key,
            "chip": entry,
            "size": Vector2i(int(area.x), int(area.y)),
            "paint": Callable(entry, "paint_preview"),
        })
    if requests.is_empty():
        _baking = false
        return
    var baked: Dictionary = await PreviewBaker.bake_all(self, requests)
    _baking = false
    # 无头环境（测试）不烘：那边读回来是空图，套上去反而会把预览抹⟦掉⟧⟪\n⟫    if baked.is_empty() or not is_inside_tree():
        return
    for request in requests:
        var key := str(request["key"])
        if not baked.has(key):
            continue
        _preview_cache[key] = baked[key]
        var entry := request["chip"] as SkinChip
        # 烘的这几帧里尺寸若又变了，这一张就作废（下一轮会按新尺寸重烘），
        # 硬贴上去只会是一张被拉伸的图
        if is_instance_valid(entry) and entry.preview_size() == Vector2(request["size"]):
            entry.preview_texture = baked[key]

## 皮肤列表可用的横向宽度：滚动视口宽度
func _skin_area_width() -> float:
    var scroll := node_named("Scroll") as ScrollContainer
    if scroll == null or scroll.size.x <= 1.0:
        return 0.0
    return maxf(scroll.size.x, 0.0)

func _fill_chips(container: Container, list: Array, cache: Array) -> void:
    if container == null or list.is_empty():
        return
    for i in list.size():
        var chip: SkinChip
        if i < cache.size():
            chip = cache[i]
        else:
            chip = SkinChip.new()
            chip.chip_pressed.connect(_on_skin_chip_pressed)
            container.add_child(chip)
            cache.append(chip)
        var entry: Dictionary = list[i]
        chip.setup(entry, bool(entry.get("owned", false)), bool(entry.get("equipped", false)))

func _make_settings_row(definition: Dictionary) -> SettingsRow:
    var key := str(definition.get("key", ""))
    if key.is_empty():
        return null
    var kind_name := str(definition.get("kind", "switch"))
    var kind := SettingsRow.Kind.SWITCH
    if kind_name == "stepper":
        kind = SettingsRow.Kind.STEPPER
    elif kind_name == "action":
        kind = SettingsRow.Kind.ACTION
    var row := SettingsRow.create(key, str(definition.get("title", "")), kind,
        _settings_icon(key, kind), str(definition.get("desc", "")))
    row.toggled.connect(_on_setting_toggled)
    row.stepped.connect(_on_setting_stepped)
    row.acted.connect(_on_setting_action)
    if kind == SettingsRow.Kind.ACTION:
        row.set_action_text(str(definition.get("action", "确定")), key != "logout")
    # 同一个键可能出现在多个分类里（音乐开关既在基础设置也在音效设置），
    # 因此用一个数组存同键的所有行，刷新时一起回填⟦。⟧⟪\n⟫    if not _settings_rows.has(key):
        _settings_rows[key] = []
    (_settings_rows[key] as Array).append(row)
    return row

func _refresh_achievements(_snapshot: Dictionary) -> void:
    if _achievement_list == null:
        return
    for child in _achievement_list.get_children():
        _achievement_list.remove_child(child)
        child.queue_free()
    var data: Dictionary = {}
    if app != null:
        data = app.achievement_snapshot()
    var items: Array = data.get("items", [])
    var unlocked_count := 0
    var last_group := ""
    for raw in items:
        var item: Dictionary = raw
        if bool(item.get("completed", false)):
            unlocked_count += 1
        var group := str(item.get("group", "成就"))
        if group != last_group:
            last_group = group
            _achievement_list.add_child(_make_achievement_group_title(group))
        _achievement_list.add_child(_make_achievement_row(item))
    if _achievement_total_bar != null:
        _achievement_total_bar.max_value = maxi(1, items.size())
        _achievement_total_bar.value = unlocked_count
    if _achievement_summary != null:
        _achievement_summary.text = "成就之路 · 已解⟦锁⟧⟪␠⟫%d / %d · 持续解锁丰厚奖励" % [unlocked_count, items.size()]

func _rebuild_messages(snapshot: Dictionary) -> void:
    if _message_list == null:
        return
    for child in _message_list.get_children():
        _message_list.remove_child(child)
        child.queue_free()
    var mail: Dictionary = {}
    if app != null:
        mail = app.mail_snapshot()
    var items: Array = []
    if bool(mail.get("has_mail", false)):
        items = mail.get("items", [])
    else:
        # 离线 / 邮件不可用：退回本地提示，保证界面不空
        items = _collect_messages(snapshot)
    var shown := 0
    for raw in items:
        var message: Dictionary = raw
        if _mail_category_of(message) != _mail_category:
            continue
        _message_list.add_child(_make_message_row(message))
        shown += 1
    if shown == 0:
        _message_list.add_child(ThemeFactory.label(
            "%s暂时没有新消⟦息⟧⟪"⟫ % str(MAIL_CATEGORIES[_mail_category]), 16, ThemeFactory.TEXT_FAINT))

## 改名行默认收起，点头像右下角的编辑角标才展开
func _build_rename_row() -> Control:
    _rename_row = UiKit.row(8)
    _rename_row.name = "RenameRow"
    _rename_row.visible = false
    _rename_edit = ThemeFactory.line_edit("新的昵称", 17)
    _rename_edit.name = "RenameEdit"
    _rename_edit.max_length = AppViewModel.NICKNAME_MAX
    _rename_edit.custom_minimum_size = Vector2(0, 52)
    _rename_edit.size_flags_horizontal = Control.SIZE_EXPAND_FILL
    _rename_row.add_child(_rename_edit)
    var save := ThemeFactory.gold_button("保存", 16)
    save.name = "RenameSaveButton"
    save.custom_minimum_size = Vector2(78, 52)
    save.pressed.connect(_on_rename_saved)
    _rename_row.add_child(save)
    var cancel := _outline_button("取消", 16)
    cancel.name = "RenameCancelButton"
    cancel.custom_minimum_size = Vector2(78, 52)
    cancel.pressed.connect(_on_rename_cancel)
    _rename_row.add_child(cancel)
    return _rename_row

func _on_rename_pressed() -> void:
    if _rename_row == null:
        return
    _rename_row.visible = true
    if _rename_edit != null:
        _rename_edit.text = _nickname_label.text
        _rename_edit.grab_focus()

func _on_rename_saved() -> void:
    if app == null or _rename_edit == null:
        return
    if app.rename(_rename_edit.text):
        _rename_row.visible = false

func _on_rename_cancel() -> void:
    if _rename_row != null:
        _rename_row.visible = false

func _on_logout_pressed() -> void:
    if app == null:
        return
    app.logout()

## 「复⟦制⟧⟪␠⟫ID」：写进系统剪贴板，并给一⟦次⟧⟪␠⟫toast 反馈
func _on_copy_id_pressed() -> void:
    if _id_label == null:
        return
    var text := _id_label.text.replace("ID: ", "").strip_edges()
    if text.is_empty() or text == "--":
        return
    DisplayServer.clipboard_set(text)
    if app != null:
        app.toast.emit("ID 已复制：%s" % text)
